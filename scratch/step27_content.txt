33: // ─── Douglas-Peucker Simplification ──────────────────────────────────────────
34: function simplifyPath(points, tolerance) {
35:   if (points.length <= 2) return points;
36:   let maxSqDist = 0;
37:   let index = 0;
38:   const end = points.length - 1;
39:   for (let i = 1; i < end; i++) {
40:     const sqDist = getSqSegDist(points[i], points[0], points[end]);
41:     if (sqDist > maxSqDist) {
42:       index = i;
43:       maxSqDist = sqDist;
44:     }
45:   }
46:   if (maxSqDist > tolerance * tolerance) {
47:     const results1 = simplifyPath(points.slice(0, index + 1), tolerance);
48:     const results2 = simplifyPath(points.slice(index), tolerance);
49:     return results1.slice(0, results1.length - 1).concat(results2);
50:   }
51:   return [points[0], points[end]];
52: }

54: function getSqSegDist(p, p1, p2) {
55:   let x = p1[0], y = p1[1];
56:   let dx = p2[0] - x, dy = p2[1] - y;
57:   if (dx !== 0 || dy !== 0) {
58:     let t = ((p[0] - x) * dx + (p[1] - y) * dy) / (dx * dx + dy * dy);
59:     if (t > 1) {
60:       x = p2[0];
61:       y = p2[1];
62:     } else if (t > 0) {
63:       x += dx * t;
64:       y += dy * t;
65:     }
66:   }
67:   dx = p[0] - x;
68:   dy = p[1] - y;
69:   return dx * dx + dy * dy;
70: }
